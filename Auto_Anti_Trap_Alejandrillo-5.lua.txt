local function findTool(name)
	if not name then return nil end
	local backpack = player:FindFirstChildOfClass("Backpack")
	if backpack then
		local tool = backpack:FindFirstChild(name)
		if tool and tool:IsA("Tool") then return tool end
	end
	local character = player.Character
	if character then
		local tool = character:FindFirstChild(name)
		if tool and tool:IsA("Tool") then return tool end
	end
	return nil
end

local function equipAndInstantUnequip()
	if not enabled or not selectedToolName then return end
	local now = os.clock()
	if now - lastEquip < COOLDOWN then return end
	local character = player.Character
	if not character then return end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return end
	local tool = findTool(selectedToolName)
	if not tool then return end
	lastEquip = now
	pcall(function() humanoid:EquipTool(tool) end)
	task.spawn(function()
		pcall(function() tool:Activate() end)
		pcall(function() humanoid:UnequipTools() end)
	end)
end

local function getTools()
	local tools = {}
	local backpack = player:FindFirstChildOfClass("Backpack")
	if backpack then
		for _,obj in ipairs(backpack:GetChildren()) do
			if obj:IsA("Tool") then tools[obj.Name] = true end
		end
	end
	local character = player.Character
	if character then
		for _,obj in ipairs(character:GetChildren()) do
			if obj:IsA("Tool") then tools[obj.Name] = true end
		end
	end
	return tools
end

local function refresh()
	for _,obj in ipairs(list:GetChildren()) do
		if obj:IsA("TextButton") then obj:Destroy() end
	end
	local tools = getTools()
	for name in pairs(tools) do
		local button = Instance.new("TextButton")
		button.Size = UDim2.new(1,-8,0,28)
		button.BackgroundColor3 = Color3.fromRGB(32,32,40)
		button.BorderSizePixel = 0
		button.Text = name
		button.TextColor3 = Color3.fromRGB(220,220,235)
		button.TextScaled = true
		button.Font = Enum.Font.Gotham
		button.Parent = list
		local c = Instance.new("UICorner")
		c.CornerRadius = UDim.new(0,6)
		c.Parent = button
		local s = Instance.new("UIStroke")
		s.Thickness = 1
		s.Transparency = 0.4
		s.Parent = button
		if selectedToolName == name then
			button.BackgroundColor3 = Color3.fromRGB(45,45,65)
		end
		button.MouseButton1Click:Connect(function()
			selectedToolName = name
			selectedLabel.Text = name
			for _,other in ipairs(list:GetChildren()) do
				if other:IsA("TextButton") then
					other.BackgroundColor3 = Color3.fromRGB(32,32,40)
				end
			end
			button.BackgroundColor3 = Color3.fromRGB(45,45,65)
		end)
	end
	list.CanvasSize = UDim2.new(0,0,0,layout.AbsoluteContentSize.Y + 5)
end

refresh()

		refresh()

local dragging = false
local dragStart
local startPos

local function updateDrag(input)
	local delta = input.Position - dragStart
	main.Position = UDim2.new(startPos.X.Scale,startPos.X.Offset + delta.X,startPos.Y.Scale,startPos.Y.Offset + delta.Y)
	byLabel.Position = UDim2.new(main.Position.X.Scale,main.Position.X.Offset,main.Position.Y.Scale,main.Position.Y.Offset + main.AbsoluteSize.Y + 3)
end

title.InputBegan:Connect(function(input)
	if locked then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = main.Position
	end
end)

title.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and not locked then
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			updateDrag(input)
		end
	end
end)

toggle.MouseButton1Click:Connect(function()
	enabled = not enabled
	stuckTime = 0
	local character = player.Character
	if character then
		local root = character:FindFirstChild("HumanoidRootPart")
		if root then lastPosition = root.Position end
	end
	toggle.Text = enabled and "ON" or "OFF"
end)

RunService.Heartbeat:Connect(function(deltaTime)
	if not enabled or not selectedToolName then return end
	local character = player.Character
	if not character then return end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart")
	if not humanoid or not root then return end
	if not lastPosition then
		lastPosition = root.Position
		return
	end
	local tryingToMove = humanoid.MoveDirection.Magnitude > 0.05
	local distance = (root.Position - lastPosition).Magnitude
	local actuallyMoving = distance > MOVEMENT_THRESHOLD
	if tryingToMove and not actuallyMoving then
		stuckTime += deltaTime
	else
		stuckTime = 0
	end
	lastPosition = root.Position
	if stuckTime >= CHECK_TIME then
		stuckTime = 0
		task.spawn(equipAndInstantUnequip)
	end
end)

local rgbTime = 0
RunService.RenderStepped:Connect(function(deltaTime)
	rgbTime += deltaTime * 0.25
	local rainbow = Color3.fromHSV(rgbTime % 1,0.85,1)
	stroke.Color = rainbow
	titleStroke.Color = rainbow
	arrowStroke.Color = rainbow
	lockStroke.Color = rainbow
	toggleStroke.Color = rainbow
	objectLabel.TextColor3 = Color3.fromHSV((rgbTime + 0.15) % 1,0.65,1)
end)

player.CharacterAdded:Connect(function()
	lastPosition = nil
	stuckTime = 0
	task.wait(0.25)
	refresh()
end)
