local CONFIG_FILE = "ShiftLockConfig.json"
local HttpService = game:GetService("HttpService")

local function loadConfig()
    local ok, result = pcall(function()
        if isfile and readfile and isfile(CONFIG_FILE) then
            return HttpService:JSONDecode(readfile(CONFIG_FILE))
        end
    end)
    if ok and type(result) == "table" then return result end
    return {}
end

local function saveConfig(data)
    pcall(function()
        if writefile then
            writefile(CONFIG_FILE, HttpService:JSONEncode(data))
        end
    end)
end

local config = loadConfig()

-- ANTI DIE
local antiDieActive = true
local function protectChar(char)
    local hum = char:WaitForChild("Humanoid", 5)
    if not hum then return end
    pcall(function() hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false) end)
    pcall(function() hum.MaxHealth = math.huge end)
    local hb
    hb = RunService.Heartbeat:Connect(function()
        if not antiDieActive or hum.Parent ~= char then
            hb:Disconnect()
            return
        end
        pcall(function() hum.Health = hum.MaxHealth end)
    end)
    bind(hb)
end
if LP.Character then task.spawn(protectChar, LP.Character) end
bind(LP.CharacterAdded:Connect(function(c) task.wait(0.05) protectChar(c) end))
print("[2] Anti die listo")

local function updateUI()
    if shiftLock then
        statusText.Text = "SHIFT LOCK ACTIVADO"
        actionBtn.Text = "DESACTIVAR SHIFT LOCK"
        indicator.BackgroundColor3 = COLOR_ACCENT
        actionStroke.Color = COLOR_ACCENT
        statusStroke.Color = COLOR_ACCENT
    else
        statusText.Text = "SHIFT LOCK DESACTIVADO"
        actionBtn.Text = "ACTIVAR SHIFT LOCK"
        indicator.BackgroundColor3 = COLOR_OFF
        actionStroke.Color = COLOR_BORDER
        statusStroke.Color = COLOR_BORDER
    end
end

local function updateBindUI()
    bindBtn.Text = "BIND  •  " .. keyLabel()
end

local function toggleShiftLock()
    if keyCooldown then return end
    keyCooldown = true
    shiftLock = not shiftLock
    if not shiftLock then restore() end
    updateUI()
    print("Shift lock:", shiftLock)
    task.delay(0.15, function() keyCooldown = false end)
end

-- Botones de la interfaz
actionBtn.MouseButton1Click:Connect(toggleShiftLock)

bindBtn.MouseButton1Click:Connect(function()
    if listening then
        listening = false
        bindBtn.Text = "BIND  •  " .. keyLabel()
        bindBtn.TextColor3 = COLOR_TEXT
        return
    end
    listening = true
    bindBtn.Text = "BIND  •  ..."
    bindBtn.TextColor3 = Color3.fromRGB(255, 190, 40)
    task.delay(6, function()
        if listening then
            listening = false
            bindBtn.Text = "BIND  •  " .. keyLabel()
            bindBtn.TextColor3 = COLOR_TEXT
        end
    end)
end)

-- Render Step para el Shift Lock
RunService:BindToRenderStep("RivalShiftLock", Enum.RenderPriority.Camera.Value + 1, function()
    if not shiftLock then return end
    local hum, root = getParts()
    local cam = workspace.CurrentCamera
    if not (hum and root and cam) then return end
    UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
    hum.AutoRotate = false
    hum.CameraOffset = OFFSET_CORRECTO
    local look = cam.CFrame.LookVector
    local flat = Vector3.new(look.X, 0, look.Z)
    if flat.Magnitude > 0.001 then
        root.CFrame = CFrame.new(root.Position, root.Position + flat)
    end
end)
print("[3] Shift lock listo")

-- KEYBINDS: mando (configurable) y teclado (Shift+F)
bind(UIS.InputBegan:Connect(function(input)
    if listening then
        if input.UserInputType.Name:find("Gamepad") then
            gamepadKey = input.KeyCode
            listening = false
            config.gamepadKey = gamepadKey.Name
            saveConfig(config) -- guarda la nueva keybind
            updateBindUI()
            bindBtn.TextColor3 = COLOR_TEXT
            print("Botón del mando guardado:", gamepadKey.Name)
        end
        return
    end

    if input.KeyCode == gamepadKey then
        toggleShiftLock()
        return
    end
    if input.KeyCode == Enum.KeyCode.F and not UIS:GetFocusedTextBox() then
        if UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.RightShift) then
            toggleShiftLock()
        end
    end
end))

print("[4] GUI creado")

if getgenv then
    getgenv().AntiDieShiftCleanup = function()
        shiftLock = false
        antiDieActive = false
        restore()
        pcall(function() RunService:UnbindFromRenderStep("RivalShiftLock") end)
        pcall(function() gui:Destroy() end)
        for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    end
end
